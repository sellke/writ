from dataclasses import dataclass

@dataclass
class SpecRow:
    name: str
    done: int
    total: int
    state: str
    blockers: list[str]

COLS = ("SPEC", "STORIES", "STATE")
WIDTHS = (32, 9, 12)  # fixed; long names truncate

def render(rows: list[SpecRow]) -> str:
    lines = [fmt(COLS)]
    lines.append("-" * sum(WIDTHS))
    notes = []
    for r in rows:
        mark = ""
        if r.blockers:
            notes.extend(r.blockers)
            mark = f" [{len(notes)}]"
        lines.append(fmt((
            r.name + mark,
            f"{r.done}/{r.total}",
            r.state,
        )))
    for i, n in enumerate(notes, 1):
        lines.append(f"[{i}] {n}")
    return "\n".join(lines)

def fmt(cells) -> str:
    return "".join(
        str(c)[:w].ljust(w)
        for c, w in zip(cells, WIDTHS)
    )
