def summary_line(spec):
    stories = load_stories(spec.path)
    done = sum(s.status == "done" for s in stories)
    blocked = sum(s.status == "blocked" for s in stories)
    line = f"{spec.slug}: {done}/{len(stories)} done"
    if blocked:
        line += f", {blocked} blocked"
    return line
